-- Trigger pour créer automatiquement un profil agent lors de l'inscription
-- 1. Création de la fonction qui insère le profil agent
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS trigger AS $$
BEGIN
  INSERT INTO public.agents (id, nom, email, role)
  VALUES (new.id, split_part(new.email, '@', 1), new.email, 'AGENT');
  RETURN new;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 2. Création du trigger qui appelle la fonction après chaque insertion dans auth.users
DROP TRIGGER IF EXISTS on_auth_user_created ON auth.users;
CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE PROCEDURE public.handle_new_user();
